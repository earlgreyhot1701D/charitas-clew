# ANTIGRAVITY MASTER DIRECTIVE: CHARITAS CLEW

## 1. THE STORY & PRD
**Mission (DEV Weekend Challenge - Generosity):** 
Charitas Clew is a mobile-first, zero-judgment public rights engine. It ingests intimidating bureaucratic notices, deconstructs the legal threats using Gemini Flash, and outputs protected rights and an actionable caseworker script. 

**Product Requirements:**
*   **Tone:** Dignified sanctuary. No alarming terms.
*   **Aesthetics:** "Deep Oceanic Neo-Editorial." Tactile, blocky, mobile-first, using cool, saturated, calming colors (Royal Azure, Vibrant Violet, Deep Teal).
*   **Safety & EOL:** Must include a Graceful End-of-Life (EOL) fallback. If the backend fails or the project sunsets, the UI must provide static links to real-world legal aid.

## 2. ARCHITECTURE
*   **Stack:** 100% Google Stack. Node.js backend (Express) + `@google/genai` SDK + Vanilla HTML/CSS/JS frontend. 
*   **Deployment:** Firebase Hosting (Frontend) + Google Cloud Run (Backend via Docker).

D. server.js
JavaScript
import express from 'express';
import { GoogleGenAI } from '@google/genai';
import path from 'path';
import { fileURLToPath } from 'url';

const __dirname = path.dirname(fileURLToPath(import.meta.url));
const app = express();
app.use(express.json());
app.use(express.static(path.join(__dirname, 'public')));

const ai = new GoogleGenAI(); // Requires GEMINI_API_KEY in environment

app.post('/api/deconstruct', async (req, res) => {
  try {
    const { text } = req.body;
    
    const response = await ai.models.generateContent({
      model: 'gemini-1.5-flash',
      contents: `Deconstruct this bureaucratic notice: ${text}`,
      config: {
        systemInstruction: "You are Charitas Clew, an impartial public legal advocate. Deconstruct bureaucratic notices into clear, dignified language. Formulate 2-3 simple next steps. Draft a respectful formal English note the user can hand to a clerk.",
        responseMimeType: "application/json",
        responseSchema: {
          type: "OBJECT",
          properties: {
            actualMeaning: { type: "STRING" },
            hasDeadline: { type: "BOOLEAN" },
            deadlineDate: { type: "STRING" },
            deadlineContext: { type: "STRING" },
            actionSteps: {
              type: "ARRAY",
              items: {
                type: "OBJECT",
                properties: {
                  title: { type: "STRING" },
                  description: { type: "STRING" }
                }
              }
            },
            advocateScript: { type: "STRING" }
          },
          required: ["actualMeaning", "hasDeadline", "actionSteps", "advocateScript"]
        }
      }
    });

    res.json(JSON.parse(response.text));
  } catch (error) {
    console.error("API Error:", error);
    res.status(500).json({ error: 'Failed to process document' });
  }
});

const PORT = process.env.PORT || 8080;
app.listen(PORT, () => console.log(`Charitas Clew engine running on port ${PORT}`));

document.getElementById('btn-copy').onclick = async () => {
  await navigator.clipboard.writeText(document.getElementById('script-text').innerText);
  document.getElementById('btn-copy').innerText = "Copied!";
  setTimeout(() => document.getElementById('btn-copy').innerText = "📋 Copy Text", 1500);
};

// Graceful EOL & Failure Fallback
function triggerEOL() {
  document.getElementById('eol-notice').classList.add('active');
  document.getElementById('btn-run').disabled = true;
  document.getElementById('btn-run').innerText = 'System Offline';
}

// Sunset check (e.g., Jan 1, 2027)
if (new Date() > new Date('2027-01-01')) { triggerEOL(); }

document.getElementById('btn-run').onclick = async () => {
  const text = document.getElementById('notice-input').value;
  if (!text) return alert("Please enter text.");
  
  const btn = document.getElementById('btn-run');
  btn.textContent = 'Processing...';
  btn.style.transform = 'translate(2px, 2px)';
  btn.style.boxShadow = '2px 2px 0px var(--ink)';
  
  try {
    const response = await fetch('/api/deconstruct', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ text })
    });
    
    if (!response.ok) throw new Error("API Failure");
    
    const data = await response.json();
    document.getElementById('meaning-text').innerText = data.actualMeaning;
    document.getElementById('deadline-val').innerText = data.hasDeadline ? data.deadlineDate : 'NONE';
    document.getElementById('deadline-context').innerText = data.hasDeadline ? data.deadlineContext : 'No immediate action required.';
    document.getElementById('script-text').innerText = data.advocateScript;
    
    document.getElementById('action-steps-container').innerHTML = data.actionSteps.map(step => 
      `<div class="step"><strong style="color: var(--ink);">\${step.title}</strong><br><span style="font-size: 0.9rem;">\${step.description}</span></div>`
    ).join('');
    
  } catch (e) {
    console.error(e);
    triggerEOL(); // Fail gracefully to human resources
  } finally {
    btn.textContent = 'Decompress Notice ↳';
    btn.style.transform = 'none';
    btn.style.boxShadow = 'var(--shadow-block)';
  }
};
